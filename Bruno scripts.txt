test("Response returns the requested plugin", function () {
  const body = res.getBody();
  expect(body.every((segment) => segment.pluginId === bru.getEnvVar("englishPluginId"))).to.be.true;
});

test("Response returns the created segments for the requested group", function () {
  const body = res.getBody();
  expect(body).to.be.an("array");
  expect(body.length).to.equal(3);

  body.forEach((segment) => {
    expect(segment.segmentId).to.exist;
    expect(segment.groupId).to.equal("core-quantities-english");
  });
});

test("Response includes the submitted setting values", function () {
  const body = res.getBody();
  body.forEach((segment) => {
    const values = segment.values;
    expect(values.maxTorque).to.equal(15.0);
    expect(values.maxWob).to.equal(30.0);
    expect(values.standLength).to.equal(90.0);
  });
});

test("Segment depth ranges are contiguous", function () {
  const body = res.getBody();
  const first = body.find((segment) => segment.startDepth === 0.0);
  const second = body.find((segment) => segment.startDepth === 100.0);
  const third = body.find((segment) => segment.startDepth === 200.0);

  expect(first, "segment starting at depth 0").to.exist;
  expect(second, "segment starting at depth 100").to.exist;
  expect(third, "segment starting at depth 200").to.exist;

  expect(first.endDepth).to.equal(100.0);
  expect(second.endDepth).to.equal(200.0);
  expect(third.endDepth).to.equal(null);
});
